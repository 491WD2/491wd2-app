/* Regenerate when renderPreview changes: see docs/ui-variant-inventory.md (node extract script in repo history). */
function readStringSetting(item: ExportedCanvasItem, key: string, fallback: string): string {
  const v = item.settings?.[key];
  return typeof v === "string" ? v : fallback;
}

function readStringSettingNonEmpty(item: ExportedCanvasItem, key: string, fallback: string): string {
  const v = item.settings?.[key];
  if (typeof v === "string" && v.trim() !== "") return v;
  return fallback;
}

function readNumberSetting(item: ExportedCanvasItem, key: string, fallback: number): number {
  const v = item.settings?.[key];
  if (typeof v === "number" && Number.isFinite(v)) return v;
  if (typeof v === "string" && v.trim() !== "") {
    const n = Number(v);
    if (!Number.isNaN(n)) return n;
  }
  return fallback;
}

function readBooleanSetting(item: ExportedCanvasItem, key: string, fallback: boolean): boolean {
  const v = item.settings?.[key];
  return typeof v === "boolean" ? v : fallback;
}

function readStringListSetting(item: ExportedCanvasItem, key: string, fallback: string[]): string[] {
  const v = item.settings?.[key];
  if (Array.isArray(v) && v.length > 0) return v;
  return fallback;
}

function cn(...parts: Array<string | false | null | undefined>): string {
  return parts.filter(Boolean).join(" ");
}

const VARIANT_OPTIONS = __VARIANT_JSON__ as Record<string, readonly string[]>;

function cssSafeVariant(v: string): string {
  const s = v.replace(/[^a-z0-9]+/gi, "-").replace(/^-+|-+$/g, "").toLowerCase();
  if (!s) return "default";
  if (/^[0-9]/.test(s)) return `r-${s}`;
  return s;
}

function normalizeVariant(definitionId: string, value: unknown): string {
  const list = VARIANT_OPTIONS[definitionId];
  if (!list || !list.length) return "default";
  if (typeof value === "string" && (list as readonly string[]).includes(value)) return value;
  return list[0];
}

function readVariant(item: ExportedCanvasItem, definitionId: string): string {
  return normalizeVariant(definitionId, readStringSetting(item, "variant", ""));
}

          <aside>
            <h3>{item.title}</h3>
            <p>{item.body}</p>

      const pct = Math.min(100, Math.max(0, readNumberSetting(item, "percent", 72)));

      const vid = readVariant(item, "tabs");

            <p>{item.body}</p>
          </>,
        );
      }

          {createElement(tag, null, item.title)}
